---
title: Dark mode
description: Switch light, dark and high-contrast themes from application state.
---

The application owns the mode and passes a resolved theme to `ThemeProvider`.

```tsx
import { useState } from "react";
import { getTheme, type ThemeMode } from "@gpuixui/tokens";
import { ThemeProvider } from "@gpuixui/tokens/react";

export function Root() {
  const [mode, setMode] = useState<ThemeMode>("dark");
  const [contrast, setContrast] = useState<"default" | "high">("default");

  return (
    <ThemeProvider theme={getTheme(mode, contrast)}>
      <Workspace onToggleMode={() => setMode(mode === "dark" ? "light" : "dark")} />
    </ThemeProvider>
  );
}
```

## Persist the choice

```tsx
const [mode, setMode] = useState<ThemeMode>(() => settings.themeMode ?? "dark");
```

## High contrast

```tsx
const theme = getTheme(mode, accessible ? "high" : "default");
```

## Platform typography

```tsx
import { getTheme, platformTheme } from "@gpuixui/tokens";

const theme = platformTheme(getTheme(mode), "win32");
```

A high-contrast base keeps its own colours.

## Usage

```tsx
import { Duration } from "effect";
import { Button } from "@gpuixui/primitives";
import {
  ToastProvider,
  ToastViewport,
  useToast,
} from "@gpuixui/components/toast";

const PublishBar = () => {
  const toast = useToast();

  return (
    <Button
      onActivate={() =>
        toast.push(
          {
            title: "Changes published",
            description: "The site rebuilds in about a minute.",
            tone: "success",
            action: { label: "Undo" },
            duration: Duration.seconds(8),
          },
          { onAction: () => restore() },
        )
      }
    >
      Publish changes
    </Button>
  );
};

export const App = () => (
  <ToastProvider>
    <PublishBar />
    <ToastViewport />
  </ToastProvider>
);
```

## API reference

| Part | Prop | Type | Default |
| ---- | ---- | ---- | ------- |
| `ToastProvider` | `max` | `number` — toasts kept before the oldest is dropped | `3` |
| `ToastProvider` | `defaultDuration` | `Duration.Input` | 5 seconds |
| `ToastProvider` | `children` | the tree that may push | — |
| `ToastViewport` | `children` | an optional control before the stack | — |
| `ToastViewport` | `className` | `string` — joined after the recipe | — |
| `useToast` | `toasts` | `ReadonlyArray<Toast>` — oldest first | — |
| `useToast` | `push` | `(draft, options?) => void` — `title` required; `description`, `tone`, `duration`, `action` optional | — |
| `useToast` | `dismiss` | `(id: string) => void` | — |
| `useToast` | `dismissAll` | `() => void` | — |
| `useToast` | `pause` | `(id: string) => void` — stop the clock | — |
| `useToast` | `resume` | `(id: string) => void` — restart with the remaining time | — |
| `useToast` | `actionFor` | `(id: string) => Option<() => void>` — the queued handler | — |
| `ToastInput` | `title` | `string` — non-empty; empty input is dropped | required |
| `ToastInput` | `description` | `string` | — |
| `ToastInput` | `tone` | `"info" \| "success" \| "warning" \| "danger"` | `"info"` |
| `ToastInput` | `duration` | `Duration.Input` | the provider's default |
| `ToastInput` | `action` | `{ label }` — the handler travels beside the draft | — |

## Limitations

- The viewport is `role="status"` with `aria-label="Notifications"`; each card's
  dismiss control is a labelled button, and **`@gpuix/native@0.11.0` publishes no
  `aria-live` or `aria-atomic`**, so a status region is the supported
  announcement.
- **A dismissed card is a ghost, not a target.** It stays painted for its
  **180 ms** exit with the content it was dismissed with, while its own action
  and dismiss controls leave the window's Tab order for that window: the runtime
  decides the order element by element, so a control the library paints has to
  drop its own stop. A control you compose inside a card is your content and
  keeps its stop until the runtime publishes an inherited `inert`.
- **The stack keeps 3 cards by default** (`max`) and the next push expires the
  oldest, so the store is a queue rather than a history: a notification centre
  that must list what expired owns that history itself. **No progress indicator**
  for the remaining time.

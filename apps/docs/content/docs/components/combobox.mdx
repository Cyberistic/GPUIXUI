## Usage

```tsx
import { useState } from "react";
import {
  Combobox,
  ComboboxContent,
  ComboboxEmpty,
  ComboboxInput,
  ComboboxList,
  ComboboxTrigger,
} from "@gpuixui/components/combobox";

const people = [
  {
    value: "ada",
    label: "Ada Lovelace",
    description: "Engineering · compilers",
  },
  {
    value: "grace",
    label: "Grace Hopper",
    description: "Platform · releases",
  },
];

export const ReviewerField = () => {
  const [reviewer, setReviewer] = useState<string | null>("ada");

  return (
    <Combobox
      onValueChange={(value) =>
        setReviewer(typeof value === "string" ? value : null)
      }
      options={people}
      value={reviewer ?? undefined}
    >
      <ComboboxTrigger label="Reviewer" placeholder="Choose a reviewer…" />
      <ComboboxContent>
        <ComboboxInput label="Search reviewers" />
        <ComboboxList label="Reviewers" />
        <ComboboxEmpty />
      </ComboboxContent>
    </Combobox>
  );
};
```

## API reference

### Combobox

| Prop | Type | Default |
| ---- | ---- | ------- |
| `options` | `ReadonlyArray<ComboboxOption>` — `{ value, label, description?, disabled? }` | required |
| `value` | `string \| ReadonlyArray<string>` — controlled | — |
| `defaultValue` | `string \| ReadonlyArray<string>` — initial selection when uncontrolled | — |
| `onValueChange` | `(value: string \| null \| ReadonlyArray<string>) => void` | — |
| `multiple` | `boolean` | `false` |
| `disabled` | `boolean` | `false` |
| `onQueryChange` | `(query: string) => void` — every search the popup runs, including the empty one it opens with | — |
| `shouldFilter` | `boolean` — rank `options` locally; `false` keeps the caller's order | `true` |
| `loading` | `boolean` — the caller's fetch is in flight; an empty list says `Searching…` | `false` |
| `onCreate` | `(query: string) => void` — single mode only; the caller adds the option | — |
| `createLabel` | `(query: string) => string` | `Create "…"` |

### ComboboxTrigger

| Prop | Type | Default |
| ---- | ---- | ------- |
| `placeholder` | `string` — shown while nothing is selected | `"Select…"` |
| `label` | `string` — the trigger's accessible name | — |

### ComboboxContent

| Prop | Type | Default |
| ---- | ---- | ------- |
| `side` | `"top" \| "right" \| "bottom" \| "left"` | `"bottom"` |
| `align` | `"start" \| "center" \| "end"` | `"start"` |
| `sideOffset` | `number` — the gap to the trigger | `4` |
| `alignOffset` | `number` — added on top of the measured centre correction | `0` |

### ComboboxInput

| Prop | Type | Default |
| ---- | ---- | ------- |
| `placeholder` | `string` | `"Search…"` |
| `label` | `string` | `"Search options"` |

### ComboboxList

| Prop | Type | Default |
| ---- | ---- | ------- |
| `label` | `string` — the listbox's accessible name | `"Options"` |
| `itemStyle` | `StateStyle<ComboboxItemState>` — resolved per row | the recipe |

### ComboboxItem

| Prop | Type | Default |
| ---- | ---- | ------- |
| `value` | `string` — the label and description are read from the root's option | required |
| `style` | `StateStyle<ComboboxItemState>` — `{ selected, highlighted, disabled }` | the recipe |

### ComboboxEmpty

| Prop | Type | Default |
| ---- | ---- | ------- |
| `children` | `ReactNode` | `No matches` |

## Limitations

- The trigger and search field are named by `label`; rows report `aria-selected`.
- An empty query announces the painted hint as `aria-description`.
- **No `aria-activedescendant`** in 0.11.0: the highlighted row is visual only.

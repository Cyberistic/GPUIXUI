## Usage

```tsx
import {
  List,
  ListItem,
  ListItemDescription,
  ListItemTitle,
} from "@gpuixui/components/list";

const files = [
  { value: "readme", title: "README.md", description: "The project's front door" },
  { value: "design", title: "Design notes", description: "docs/design-notes.md" },
];

export const FileList = ({ open }: { open: (value: string) => void }) => (
  <List defaultValue="readme" onActivate={open} selectionMode="single">
    {files.map((file) => (
      <ListItem key={file.value} textValue={file.title} value={file.value}>
        <ListItemTitle>{file.title}</ListItemTitle>
        <ListItemDescription>{file.description}</ListItemDescription>
      </ListItem>
    ))}
  </List>
);
```

## API reference

### List

| Prop | Type | Default |
| ---- | ---- | ------- |
| `selectionMode` | `"none" \| "single" \| "multiple"` | `"none"` |
| `value` | `string` (single) · `ReadonlyArray<string>` (multiple) — controlled | — |
| `defaultValue` | the same shape, initial selection when uncontrolled | `""` · `[]` |
| `onValueChange` | `(value: string) => void` (single) · `(values: ReadonlyArray<string>) => void` (multiple) | — |
| `onActivate` | `(value: string) => void` — single and `none` only; absent from the multiple variant | — |
| `dense` | `boolean` — a row at `density.rowHeight` instead of `rowHeight + spacing.md` | `false` |
| `children` | parts — `ListItem` rows, one `ListSection` per group | — |
| `className` | `string` — joined after the root recipe | — |

### ListItem

| Prop | Type | Default |
| ---- | ---- | ------- |
| `value` | `string` (required) | — |
| `textValue` | `string` — the text type-ahead matches; omit and the row is not searched | — |
| `disabled` | `boolean` | `false` |
| `leading` | `ReactNode` — an icon, avatar or marker | — |
| `trailing` | `ReactNode` — metadata, a badge or an action | — |
| `children` | parts — `ListItemTitle`, `ListItemDescription` and the row's slots | — |
| `className` | `string` — joined after the row recipe | — |
| `style` | `StyleDesc` or a function of `{ active, disabled, selected }` | — |

## Limitations

- `list`/`listitem` without a mode; `listbox`/`option` with one.
- **No `aria-multiselectable`** in 0.11.0: the mode is read from each row's `aria-selected`.
- The row ring is keyboard-only: `focusVisible` paints after a key focus, not a click.

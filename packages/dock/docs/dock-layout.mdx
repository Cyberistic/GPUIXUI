---
title: Docking layouts
description: A named tree of split and tab groups, rendered from a schema
---

## Usage

```tsx
import { DockLayout, DockTab, type DockNode } from "@gpuixui/dock";
import { useState } from "react";

const initial: DockNode = {
  kind: "split",
  id: "root",
  orientation: "horizontal",
  size: 0.6,
  first: {
    kind: "tabs",
    id: "editor",
    activeTab: "code",
    tabs: [new DockTab({ id: "code", label: "Code" })],
  },
  second: {
    kind: "tabs",
    id: "preview",
    activeTab: "outline",
    tabs: [new DockTab({ id: "outline", label: "Outline" })],
  },
};

const [layout, setLayout] = useState(initial);

<DockLayout
  layout={layout}
  onLayoutChange={setLayout}
  renderTab={(id) => <Pane id={id} />}
/>;
```

## API

| Prop | Type | Default | Consequence |
| ---- | ---- | ------- | ----------- |
| `layout` | `DockNode` | — | the tree; controlled |
| `onLayoutChange` | `(layout: DockNode) => void` | — | the next tree |
| `renderTab` | `(id: string) => ReactNode` | — | one tab's content |
| `className` | `string` | — | joined after the recipe |

## Limitations

- The tree is **controlled**: every change arrives through `onLayoutChange`.
- A layout string that is not a tree is a decode failure; open the default.
- A split holds exactly two children; a tab press rewrites its group's `activeTab`.
